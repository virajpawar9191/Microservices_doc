-- ============================================================
-- PostgreSQL DDL — Employee HR System
-- Email-parsed data pipeline for AI Agent + PowerBI
-- ============================================================

CREATE TABLE employee (
  employee_id   SERIAL          PRIMARY KEY,
  email         VARCHAR(255)    NOT NULL UNIQUE,
  phone_number  VARCHAR(25),
  business_unit VARCHAR(100),
  full_name     VARCHAR(200)    NOT NULL,
  department    VARCHAR(100),
  manager       VARCHAR(200),
  status        VARCHAR(10)     NOT NULL
                  CHECK (status IN ('active', 'inactive'))
                  DEFAULT 'active',
  created_date  TIMESTAMP       NOT NULL DEFAULT NOW(),
  updated_date  TIMESTAMP       NOT NULL DEFAULT NOW()
);

-- Indexes on EMPLOYEE
CREATE UNIQUE INDEX idx_employee_email
  ON employee (email);

CREATE INDEX idx_employee_status
  ON employee (status);

CREATE INDEX idx_employee_dept
  ON employee (department);

-- ============================================================

CREATE TABLE onboarding (
  id              SERIAL          PRIMARY KEY,
  employee_id     INTEGER         NOT NULL
    REFERENCES employee (employee_id)
    ON DELETE CASCADE ON UPDATE CASCADE,
  ctoolid         INTEGER,
  sender_mail     VARCHAR(255)    NOT NULL,
  role_name       VARCHAR(150),
  jd              TEXT,
  location        VARCHAR(150),
  onboarding_date DATE,
  experience      VARCHAR(50),
  created_date    TIMESTAMP       NOT NULL DEFAULT NOW()
);

-- Indexes on ONBOARDING
CREATE INDEX idx_onboarding_emp
  ON onboarding (employee_id);

CREATE INDEX idx_onboarding_date
  ON onboarding (onboarding_date);

CREATE INDEX idx_onboarding_role
  ON onboarding (role_name);

-- ============================================================

CREATE TABLE offboarding (
  id               SERIAL          PRIMARY KEY,
  employee_id      INTEGER         NOT NULL
    REFERENCES employee (employee_id)
    ON DELETE CASCADE ON UPDATE CASCADE,
  ctoolid          INTEGER,
  sender_mail      VARCHAR(255)    NOT NULL,
  role_name        VARCHAR(150),
  offboarding_date DATE,
  created_date     TIMESTAMP       NOT NULL DEFAULT NOW()
);

-- Indexes on OFFBOARDING
CREATE INDEX idx_offboarding_emp
  ON offboarding (employee_id);

CREATE INDEX idx_offboarding_date
  ON offboarding (offboarding_date);

CREATE INDEX idx_offboarding_role
  ON offboarding (role_name);

-- ============================================================
-- Auto-update trigger: updated_date on employee
-- ============================================================
CREATE OR REPLACE FUNCTION fn_update_timestamp()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_date = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_employee_updated
  BEFORE UPDATE ON employee
  FOR EACH ROW EXECUTE FUNCTION fn_update_timestamp();

-- ============================================================
-- Useful VIEW for PowerBI DirectQuery
-- ============================================================
CREATE OR REPLACE VIEW vw_employee_lifecycle AS
SELECT
  e.employee_id,
  e.full_name,
  e.email,
  e.department,
  e.business_unit,
  e.manager,
  e.status,
  o.onboarding_date,
  o.role_name        AS onboarding_role,
  o.location         AS work_location,
  o.experience,
  off.offboarding_date,
  off.role_name      AS offboarding_role,
  CASE
    WHEN off.offboarding_date IS NOT NULL
    THEN (off.offboarding_date - o.onboarding_date)
  END                AS tenure_days
FROM employee e
LEFT JOIN onboarding  o   ON o.employee_id  = e.employee_id
LEFT JOIN offboarding off ON off.employee_id = e.employee_id;

📊 Recommended PowerBI Dashboard Architecture

Connect PowerBI to PostgreSQL via DirectQuery or Import mode. Use the vw_employee_lifecycle view as the base dataset. Create calculated measures in DAX. Recommended: 3 report pages — Onboarding, Offboarding, Executive Summary.

🟢 Onboarding KPIs — Page 1

📌 Total Onboarded (MTD / YTD)
COUNT(*) FROM onboarding GROUP BY DATE_TRUNC('month', onboarding_date)
📌 Avg Time to Onboard (Days)
AVG(o.onboarding_date - e.created_date) FROM onboarding o JOIN employee e USING(employee_id)
📌 Onboarding by Department
COUNT(*), e.department FROM onboarding o JOIN employee e USING(employee_id) GROUP BY e.department ORDER BY 1 DESC
📌 Onboarding by Location
COUNT(*), location FROM onboarding GROUP BY location ORDER BY 1 DESC
📌 Top Roles Being Hired
COUNT(*), role_name FROM onboarding GROUP BY role_name ORDER BY 1 DESC LIMIT 10
📌 Experience Distribution
COUNT(*), experience FROM onboarding GROUP BY experience

🔴 Offboarding KPIs — Page 2

📌 Total Offboarded (MTD / YTD)
COUNT(*) FROM offboarding GROUP BY DATE_TRUNC('month', offboarding_date)
📌 Attrition Rate %
-- DAX measure DIVIDE([Offboarded Count], [Active Employees], 0) * 100
📌 Avg Employee Tenure (Days)
AVG(off.offboarding_date - o.onboarding_date) FROM offboarding off JOIN onboarding o USING(employee_id)
📌 Offboarding by Department
COUNT(*), e.department FROM offboarding off JOIN employee e USING(employee_id) GROUP BY e.department
📌 Role-wise Exit Analysis
COUNT(*), role_name FROM offboarding GROUP BY role_name ORDER BY 1 DESC

📈 Executive Summary — Page 3

📌 Active vs Inactive Headcount
COUNT(*), status FROM employee GROUP BY status
📌 Headcount by Business Unit
COUNT(*), business_unit FROM employee WHERE status='active' GROUP BY business_unit
📌 Manager Span of Control
COUNT(employee_id), manager FROM employee WHERE status='active' GROUP BY manager ORDER BY 1 DESC
📌 Net Headcount (Hires − Exits)
-- DAX: Net HC per month [Onboarded MTD] - [Offboarded MTD]
📧 Step 1 — Email Ingestion

Emails arrive from HR systems (onboarding confirmations, exit notices, welcome emails). Raw email data including sender, subject, body, attachments is captured and queued for processing.

✓ SMTP / IMAP capture
✓ sender_mail extracted
✓ Subject classification
✓ Attachment parsing
🤖 Step 2 — AI Agent Processing

AI agent classifies email type (onboarding vs offboarding), performs named-entity extraction for employee name, role, department, dates, and matches to existing employee via email lookup on the EMPLOYEE table.

✓ Email classification NLP
✓ Entity extraction
✓ Employee ID resolution
✓ Confidence scoring
🗄️ Step 3 — PostgreSQL Storage

Structured data is upserted into the 3-table schema. The EMPLOYEE table acts as the master identity store (keyed by email). ONBOARDING and OFFBOARDING events are inserted as child records with FK linkage.

✓ UPSERT on email key
✓ FK integrity enforced
✓ Timestamps auto-set
✓ Trigger: updated_date
📊 Step 4 — PowerBI Dashboard

PowerBI connects via DirectQuery to PostgreSQL using the vw_employee_lifecycle view. Scheduled refresh keeps dashboards current. DAX measures calculate KPIs across Onboarding, Offboarding, and Executive pages.

✓ Onboarding trends page
✓ Attrition analytics
✓ Headcount drill-down
✓ Manager / dept slicers
